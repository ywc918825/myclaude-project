# 数据库字段结构说明

- 数据库：`server/data/talent.db`（SQLite）
- 业务表：7 张，字段合计 64 个

## 一、表总览

| 表名 | 中文说明 | 字段数 | 当前行数 | 索引数 | 外键数 |
|---|---|---:|---:|---:|---:|
| `attachments` | 附件表（证明材料） | 8 | 1 | 2 | 2 |
| `points_items` | 积分申报明细表（17 大项通用） | 19 | 3 | 3 | 2 |
| `profile_facts` | 人才画像指标事实表（5 维 18 项） | 8 | 0 | 2 | 1 |
| `role_permissions` | 角色权限关联表 | 3 | 22 | 2 | 1 |
| `roles` | 角色表 | 6 | 3 | 2 | 0 |
| `system_config` | 系统配置表 | 2 | 3 | 1 | 0 |
| `users` | 用户表 | 18 | 254 | 3 | 0 |

## 二、字段明细

### attachments（附件表（证明材料））

## 五、建表语句

### attachments

```sql
CREATE TABLE attachments (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  item_id       INTEGER NOT NULL,                -- 关联 points_items
  uploader_id   INTEGER,                         -- 上传人（关联 users）
  original_name TEXT NOT NULL,                   -- 原始文件名
  stored_name   TEXT NOT NULL,                   -- 服务器存储名（uuid+扩展名）
  mime_type     TEXT,                            -- 文件 MIME
  size          INTEGER DEFAULT 0,               -- 字节数
  created_at    TEXT NOT NULL DEFAULT (datetime('now','localtime')),
  FOREIGN KEY (item_id) REFERENCES points_items(id) ON DELETE CASCADE,
  FOREIGN KEY (uploader_id) REFERENCES users(id) ON DELETE SET NULL
)
```

### points_items

```sql
CREATE TABLE points_items (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id     INTEGER NOT NULL,
  year        INTEGER NOT NULL,
  category    INTEGER NOT NULL,            -- 1..17
  title       TEXT NOT NULL,                -- 项目/奖项/论文等名称
  -- 通用字段
  level       TEXT,                        -- 国家级/省部级/市厅级/县处级 等
  role_rank   TEXT,                        -- 第一/第二/主持/参与 等
  role_type   TEXT,                        -- 学科带头人/成员 等
  amount      REAL DEFAULT 0,              -- 资助金额/转化金额 等
  -- JSON 存储大项特殊字段（如共一作者人数、字数等）
  extras      TEXT,                        -- JSON
  -- 自评得分
  self_score  REAL NOT NULL DEFAULT 0,
  -- 附件
  evidence    TEXT,                        -- 附件路径或描述
  -- 状态
  status      TEXT NOT NULL DEFAULT 'draft', -- draft/pending/approved/rejected
  reviewer_id INTEGER,
  reviewed_at TEXT,
  review_note TEXT,
  created_at  TEXT NOT NULL DEFAULT (datetime('now','localtime')),
  updated_at  TEXT NOT NULL DEFAULT (datetime('now','localtime')), obtain_date TEXT,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (reviewer_id) REFERENCES users(id) ON DELETE SET NULL
)
```

### profile_facts

```sql
CREATE TABLE profile_facts (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id     INTEGER NOT NULL,
  fact_key    TEXT NOT NULL,                 -- 如 'papers_sci', 'patents_invention'
  fact_value  TEXT,                          -- 自由文本或数字
  fact_count  INTEGER DEFAULT 0,             -- 数量（如论文数）
  fact_score  REAL DEFAULT 0,                -- 折算得分
  year        INTEGER,
  updated_at  TEXT NOT NULL DEFAULT (datetime('now','localtime')),
  UNIQUE (user_id, fact_key, year),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
)
```

### role_permissions

```sql
CREATE TABLE role_permissions (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  role_id    INTEGER NOT NULL,
  permission TEXT NOT NULL,
  FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE,
  UNIQUE (role_id, permission)
)
```

### roles

```sql
CREATE TABLE roles (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  code        TEXT NOT NULL UNIQUE,            -- 角色标识，users.role 引用此 code
  name        TEXT NOT NULL,                   -- 角色显示名
  description TEXT,
  is_system   INTEGER NOT NULL DEFAULT 0,      -- 内置角色不可删除/改 code
  created_at  TEXT NOT NULL DEFAULT (datetime('now','localtime'))
)
```

### system_config

```sql
CREATE TABLE system_config (
  key   TEXT PRIMARY KEY,
  value TEXT
)
```

### users

```sql
CREATE TABLE users (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  username      TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  name          TEXT NOT NULL,
  department    TEXT NOT NULL,
  title         TEXT,                       -- 正高/副高/中级/管理
  talent_role   TEXT NOT NULL DEFAULT '人才', -- 人才/其他
  -- 系统角色
  role          TEXT NOT NULL DEFAULT 'user', -- admin / reviewer / user
  -- 个人信息（人才画像用）
  age           INTEGER,
  gender        TEXT,
  phone         TEXT,
  email         TEXT,
  education     TEXT,                       -- 学历：博士/硕士/本科
  major         TEXT,                       -- 专业
  work_years    INTEGER DEFAULT 0,
  bio           TEXT,
  created_at    TEXT NOT NULL DEFAULT (datetime('now','localtime')),
  updated_at    TEXT NOT NULL DEFAULT (datetime('now','localtime'))
)
```
